\section*{Parte 1: Questões Dissertativas}

% --- QUESTÃO 1 ---
\begin{tcolorbox}[colback=azulTitulo!5, colframe=azulTitulo, title=Questão 1: HTML5 Semântico e Estrutura]
    Explique por que a utilização de tags semânticas (como \texttt{<header>}, \texttt{<nav>}, \texttt{<main>} e \texttt{<article>}) é preferível ao uso genérico de \texttt{<div>}. Mencione pelo menos dois benefícios diretos (ex: SEO, acessibilidade).
    \vspace{4cm}
    \checkGabarito{Resposta esperada: Melhora o SEO (motores de busca entendem a estrutura), aumenta a acessibilidade (leitores de tela navegam melhor) e facilita a manutenção do código.}
\end{tcolorbox}

% --- QUESTÃO 2 ---
\begin{tcolorbox}[colback=azulTitulo!5, colframe=azulTitulo, title=Questão 2: Manipulação Dinâmica do DOM]
    Descreva o processo necessário para criar um novo elemento na página via JavaScript e inseri-lo em um container existente. Quais métodos do objeto \texttt{document} e de elementos do DOM são essenciais nesse fluxo?
    \vspace{4cm}
    \checkGabarito{Resposta esperada: 1. createElement() para fabricar o elemento; 2. innerText/innerHTML para dar conteúdo; 3. appendChild() ou append() para inserir no container pai.}
\end{tcolorbox}

% --- QUESTÃO 3 ---
\begin{tcolorbox}[colback=azulTitulo!5, colframe=azulTitulo, title=Questão 3: Requisições Assíncronas (Fetch API)]
    Utilizando a metáfora da "Esteira de Produção" vista em aula, descreva o que acontece em uma requisição utilizando a Fetch API, desde o pedido (\textit{fetch}) até a exibição dos dados convertidos em JSON na tela.
    \vspace{4cm}
    \checkGabarito{Resposta esperada: O fetch faz o pedido à rede; o primeiro .then() recebe a resposta bruta e converte para JSON (response.json()); o segundo .then() recebe os dados prontos para serem manipulados via DOM.}
\end{tcolorbox}

    \begin{enumerate}[label=\alph*)]
        \item O Flexbox é ideal para layouts bidimensionais (linhas e colunas simultâneas).
